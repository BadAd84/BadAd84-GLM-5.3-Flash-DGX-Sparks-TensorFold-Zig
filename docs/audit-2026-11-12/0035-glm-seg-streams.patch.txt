Subject: [PATCH] glm: TF_GLM_OVN_SEG_STREAMS, a shared decode window's KDA chains and MLA attention on side streams

Written for this series (lane KdaConcurrency, 2026-11-14); not a port. Upstream v1.2.7 runs a shared window's
per-stream work in order on the forward's stream (cuda_shared.zig: for each segment the projection-row copy and the
KDA chain, then for each segment the MLA attention); production's Python engine gets the overlap from segment kernels
(work/accept-gap/prod/forward.py 790-812 kda_segments, 1181-1233 dsa_segments). This keeps every kernel and its
arguments and moves only the stream each segment runs on.

TF_GLM_OVN_SEG_STREAMS=N (u64, a .glm row of platform/environment.zig, at most 8; 0 or 1 off, default off): with N >= 2
the engine makes N - 1 non-blocking side streams and N events at open (cuda_seg_streams.zig Streams, only with the MTP
head, which is when shared windows exist). In a shared decode window of two or more segments, after the projections
the caller's stream records a fork that each used side stream waits for; segment k runs on stream k % N (0 the
caller's); afterwards the caller waits on a join event recorded on each used side stream, before o_proj.
- KDA (cuda_seg_streams.kdaChains): each segment's copy into its state and its chain (prep, step, out, or the fused
  chain) as before, on its stream, with the three-kernel chain's q and y scratch at the segment's own rows
  (wideAt: q + row0 n 4, y + row0 n 2; Buffers already hold every row of the window).
- MLA (mlaAttends): each segment's latent write, index write, pooled keys, chunks and merge on its stream, its chunk
  partials laid after the earlier segments' (partialsAt), when mlaApart holds: every row dense, a segment within one
  attention call, no host KV pages (pks), and the partials within Buffers.partial_rows (new field, the size make()
  already computed). Otherwise, and for prompt buffers and lone segments (forWindow), the loop stays in order.
Bits: the same kernels with the same arguments on the same data in the same order within each stream; the only
changed arguments are the scratch addresses (q, y, partials), which no other segment reads. Knob off is today's launch
sequence: the in-order path passes the old scratch and the caller's stream.

Why: the profile of the recipe (docs/audit-2026-11-12/verify-gap.md, rank 0, ~/nsys/rec-p{2,4}.sqlite) launches the
KDA chain and MLA attention once per stream in order: 1.75 / 2.97 ms and 0.65 / 1.29 ms a round at 2 / 4 streams.
Bound, re-measured here from those exports (work/kda-concurrency/chainfloor.py, chainfloor.out; KDA steps are bound by
state bytes, 2 x 22 x 128 x 128 x 4 = 2.88 MB a step, 12.6 us alone = 229 GB/s of the GB10's 273, so the floor keeps
every step's measured time back to back; MLA kernels are latency bound, 6 to 66 blocks each, floor the longest
segment): KDA span 61.1 -> 47.0 us a layer at 2 streams, 102.3 -> 75.3 at 4 (x34 layers: 0.48 / 0.92 ms a round); MLA
63.2 -> 33.9, 125.3 -> 36.0 (x11: 0.32 / 0.98 ms). The GB10's own cross-stream fork, from the L2 prefetch's events in
the same captures (forklat.py), is 3.0 us median (p90 4.0) against 0.3-0.5 us between kernels on one stream: about
2 x 2.7 us a layer for a fork and a join, 45 layers 0.24 ms. Expected: up to ~0.55 ms a round at 2 streams (~1%) and
~1.6 ms at 4 (~2%), more at 6 and 8, none at 1 (a lone segment takes no stream).

+//! TF_GLM_OVN_SEG_STREAMS: a shared decode window's per-segment KDA chains and MLA attention on concurrent streams.
+
+const std = @import("std");
+const cuda = @import("cuda");
+const kda = cuda.kda;
+const fw = @import("cuda_forward.zig");
+const state = @import("cuda_state.zig");
+const kda_block = @import("cuda_kda_block.zig");
+const dsa_block = @import("cuda_dsa_block.zig");
+const latent = @import("cuda_latent.zig");
+
+/// The most streams one window's segments run on at once: the caller's and seven side streams (the row's max).
+pub const most = 8;
+
+/// The knob's stream count: 0 unset, 0 or 1 (every segment on the caller's stream, as without it).
+pub fn wanted() error{BadOvnKnob}!usize {
+    const n = try cuda.ovn.int("SEG_STREAMS", 0);
+    return if (n < 2) 0 else @intCast(n);
+}
+
+/// The engine's streams for the knob, made once at open (null: off); rank 0 logs how many a window takes.
+pub fn open(d: *const cuda.Driver, rank: usize) !?Streams {
+    const n = try wanted();
+    if (n == 0) return null;
+    if (rank == 0) std.log.info("TF_GLM_OVN_SEG_STREAMS: a shared decode window's segments on up to {d} streams", .{n});
+    return try Streams.init(d, n);
+}
+
+/// Side streams and their events; segment k of a window runs on stream k % n, stream 0 the caller's.
+pub const Streams = struct {
+    n: usize,
+    side: [most - 1]cuda.Stream = undefined,
+    joins: [most - 1]cuda.Event = undefined,
+    fork_ev: cuda.Event,
+
+    pub fn init(d: *const cuda.Driver, n: usize) !Streams {
+        if (n < 2 or n > most) return error.SegStreams;
+        var s: Streams = .{ .n = 1, .fork_ev = try cuda.Event.init(d, false) };
+        errdefer s.deinit();
+        while (s.n < n) : (s.n += 1) {
+            s.side[s.n - 1] = try cuda.Stream.init(d, true);
+            s.joins[s.n - 1] = cuda.Event.init(d, false) catch |err| {
+                s.side[s.n - 1].deinit();
+                return err;
+            };
+        }
+        return s;
+    }
+
+    /// The side streams drain first (Stream.deinit synchronizes).
+    pub fn deinit(s: *Streams) void {
+        for (s.side[0 .. s.n - 1], s.joins[0 .. s.n - 1]) |*x, *e| {
+            x.deinit();
+            e.deinit();
+        }
+        s.fork_ev.deinit();
+        s.* = undefined;
+    }
+
+    /// The stream segment k of a window runs on.
+    pub fn stream(s: *const Streams, caller: cuda.Stream, k: usize) cuda.Stream {
+        const j = k % s.n;
+        return if (j == 0) caller else s.side[j - 1];
+    }
+
+    /// The side streams a window of `segs` segments puts work on.
+    fn sides(s: *const Streams, segs: usize) usize {
+        return @min(segs, s.n) -| 1;
+    }
+
+    /// Each side stream a window uses waits for the caller's work so far: the rows its segments read.
+    pub fn fork(s: *const Streams, caller: cuda.Stream, segs: usize) !void {
+        if (s.sides(segs) == 0) return;
+        try s.fork_ev.record(caller);
+        for (s.side[0..s.sides(segs)]) |x| try x.wait(s.fork_ev);
+    }
+
+    /// The caller waits for each side stream the window used, then `res` (the segments' launches) is returned.
+    pub fn join(s: *const Streams, caller: cuda.Stream, segs: usize, res: anyerror!void) anyerror!void {
+        for (s.side[0..s.sides(segs)], s.joins[0..s.sides(segs)]) |x, e| {
+            try e.record(x);
+            try caller.wait(e);
+        }
+        return res;
+    }
+};
+
+/// The streams a window's segments take: none for a prompt buffer or a lone segment.
+pub fn forWindow(ss: ?*const Streams, prefill: bool, segs: usize) ?*const Streams {
+    if (prefill or segs < 2) return null;
+    return ss;
+}
+
+/// Segment g's rows of the window's KDA scratch at layer li: decay, gate and out rows, and the state copy it writes.
+pub fn kdaRows(s: kda_block.Scratch, g: fw.Seg, li: usize, n: usize) kda_block.Scratch {
+    var v = s;
+    v.ka += g.row0 * n * 2;
+    v.kg += g.row0 * n * 2;
+    v.saves.out += g.row0 * n * 2;
+    v.saves.state_out = g.st.recAt(1 - g.st.cur[li], li);
+    return v;
+}
+
+/// A sequence's conv window of KDA layer li.
+pub fn convAt(st: *const state.State, li: usize) u64 {
+    return st.conv + li * 3 * st.conv_c * 2;
+}
+
+/// A segment's wide q and y (the three-kernel chain's scratch) at its own rows, so no two segments share them.
+pub fn wideAt(w: kda.Wide, row0: usize, n: usize) kda.Wide {
+    return .{ .q = w.q + row0 * n * 4, .y = w.y + row0 * n * 2 };
+}
+
+/// A decode window's KDA chains at layer li: with `ss`, segment k on stream k % n at its own wide rows, else in order.
+pub fn kdaChains(d: *const cuda.Driver, ks: *const kda.Kernels, caller: cuda.Stream, ss: ?*const Streams, l: kda_block.Layer, b: *const state.Buffers, segs: []const fw.Seg, li: usize, wide: bool) !void {
+    if (ss) |x| try x.fork(caller, segs.len);
+    const res = kdaLaunch(d, ks, caller, ss, l, b, segs, li, wide);
+    if (ss) |x| return x.join(caller, segs.len, res);
+    return res;
+}
+
+fn kdaLaunch(d: *const cuda.Driver, ks: *const kda.Kernels, caller: cuda.Stream, ss: ?*const Streams, l: kda_block.Layer, b: *const state.Buffers, segs: []const fw.Seg, li: usize, wide: bool) !void {
+    const n = l.heads * kda.dk;
+    const w = l.matWidth();
+    for (segs, 0..) |g, k| {
+        const st = g.st;
+        const on = if (ss) |x| x.stream(caller, k) else caller;
+        var v = kdaRows(b.kda, g, li, n);
+        if (ss != null) v.wide = wideAt(b.kda.wide, g.row0, n);
+        v.p = st.proj + li * st.proj_layer * 2; // the window's rows stay with the layer until its commit
+        try d.check(d.api.cuMemcpyDtoDAsync_v2(v.p, b.kproj + g.row0 * w * 2, g.rows * w * 2, on.handle), "cuMemcpyDtoDAsync");
+        const kv = li * state.window_rows * st.heads * kda.dk;
+        v.saves.k = st.save_k + kv * 4;
+        v.saves.v = st.save_v + kv * 2;
+        v.saves.g = st.save_g + kv * 4;
+        v.saves.b = st.save_b + li * state.window_rows * st.heads * 4;
+        try kda_block.chainRows(ks, on, l, v, g.rows, convAt(st, li), st.recAt(st.cur[li], li), wide, null);
+    }
+}
+
+/// A segment's chunk partials after `at` partial rows of the segments before it: heads x lw floats a row, heads m, l.
+pub fn partialsAt(p: latent.Partials, at: usize, heads: usize, lw: usize) latent.Partials {
+    return .{ .po = p.po + at * heads * lw * 4, .pm = p.pm + at * heads * 4, .pl = p.pl + at * heads * 4 };
+}
+
+/// A segment's 512-key chunks: those its last row reaches, at most the buffer's.
+pub fn chunksOf(g: fw.Seg, nch: usize) usize {
+    return @min(latent.chunksFor(g.st.pos + g.rows), nch);
+}
+
+/// Whether MLA layer di's segments attend apart: dense rows in one call each, no host KV pages, partials in the room.
+pub fn mlaApart(l: dsa_block.Layer, b: *const state.Buffers, segs: []const fw.Seg, di: usize) bool {
+    var at: usize = 0;
+    for (segs) |g| {
+        if (dsa_block.sparseRows(l, g.st.pos, g.rows) or g.rows > b.part_rows or g.st.caches[di].pks != 0) return false;
+        at += chunksOf(g, b.nch) * g.rows;
+    }
+    return at <= b.partial_rows;
+}
+
+/// A window's MLA attention at layer di; with `ss` and mlaApart, segment k on stream k % n with its own partials.
+pub fn mlaAttends(k: dsa_block.Kernels, caller: cuda.Stream, ss: ?*const Streams, l: dsa_block.Layer, b: *const state.Buffers, segs: []const fw.Seg, di: usize) !void {
+    const apart = if (ss) |x| (if (mlaApart(l, b, segs, di)) x else null) else null;
+    if (apart) |x| try x.fork(caller, segs.len);
+    const res = mlaLaunch(k, caller, apart, l, b, segs, di);
+    if (apart) |x| return x.join(caller, segs.len, res);
+    return res;
+}

+/// One recorded call: a launch (kernel id, the arguments `Rec.picks` names for it), a copy, an event record or a wait.
+const Op = struct {
+    kind: enum { launch, copy, record, wait },
+    stream: usize,
+    id: usize = 0,
+    args: [4]u64 = @splat(0),
+    dst: u64 = 0,
+    src: u64 = 0,
+    bytes: usize = 0,

+const Rec = struct {
+    var ops: [256]Op = undefined;
+    var n: usize = 0;
+    var next: u64 = 0x100000000;
+
+    /// The u64 arguments kept by kernel id: prep q; step state_in, q, y, state_out; out y, out; chain out, state_out.
+    fn picks(id: usize) []const usize {
+        return switch (id) {
+            4 => &.{11},
+            5 => &.{ 1, 2, 8, 9 },
+            6 => &.{ 1, 6 },
+            1 => &.{ 17, 18 },
+            21 => &.{2}, // lat_write's cache
+            22 => &.{5}, // index_write's ik
+            23 => &.{3}, // pool_keys' pk
+            24, 25 => &.{ 5, 6, 7 }, // chunks' po, pm, pl
+            26 => &.{ 0, 1, 2, 3 }, // merge's po, pm, pl, out
+            else => &.{},
+        };
+    }
+    fn put(o: Op) void {
+        ops[n] = o;
+        n += 1;
+    }
+    fn alloc(out: *u64, bytes: usize) callconv(.c) c_int {
+        out.* = next;
+        next += std.mem.alignForward(u64, bytes, 256);
+        return 0;
+    }
+    fn free(_: u64) callconv(.c) c_int {
+        return 0;
+    }
+    fn fill(_: u64, _: u8, _: usize) callconv(.c) c_int {
+        return 0;
+    }
+    fn launch(f: cuda.abi.Function, _: c_uint, _: c_uint, _: c_uint, _: c_uint, _: c_uint, _: c_uint, _: c_uint, s: cuda.abi.Stream, args: ?[*]?*anyopaque, _: ?[*]?*anyopaque) callconv(.c) c_int {
+        const id = @intFromPtr(f);
+        var o: Op = .{ .kind = .launch, .stream = @intFromPtr(s), .id = id };
+        for (picks(id), 0..) |at, i| o.args[i] = @as(*const u64, @ptrCast(@alignCast(args.?[at].?))).*;
+        put(o);
+        return 0;
+    }
+    fn copy(dst: u64, src: u64, bytes: usize, s: cuda.abi.Stream) callconv(.c) c_int {
+        put(.{ .kind = .copy, .stream = @intFromPtr(s), .dst = dst, .src = src, .bytes = bytes });
+        return 0;
+    }
+    fn record(e: cuda.abi.Event, s: cuda.abi.Stream) callconv(.c) c_int {
+        put(.{ .kind = .record, .stream = @intFromPtr(s), .id = @intFromPtr(e) });
+        return 0;
+    }

+    const segs = segments(x, rows, &buf);
+    var total: usize = 0;
+    for (rows) |m| total += m;
+    const n = l.heads * kda.dk;
+    for (segs, 0..) |_, i| try x.start(i, KdaFill{ .li = li });
+    x.stage.clearRetainingCapacity();
+    try x.stageAt(x.b.kproj, try bf16s(x.r, total * l.matWidth(), 1.0));
+    try x.stageAt(x.b.kda.ka, try bf16s(x.r, total * n, 1.0));
+    try x.stageAt(x.b.kda.kg, try bf16s(x.r, total * n, 1.0));
+    const out_bytes = total * n * 2;
+    var want: []u8 = &.{};
+    var name: [200]u8 = undefined;
+    for ([_]Arm{ .loop, .in_order, .{ .streams = 2 }, .{ .streams = 4 }, .{ .streams = 8 } }) |arm| {
+        try x.reset(segs.len, x.b.kda.saves.out, out_bytes);
+        switch (arm) {
+            .loop => try kdaLoop(x, l, segs, li, wide),
+            .in_order => try seg.kdaChains(x.r.gpu.d, &x.ks, x.r.s, null, l, &x.b, segs, li, wide),
+            .streams => |k| try seg.kdaChains(x.r.gpu.d, &x.ks, x.r.s, x.streamsOf(k), l, &x.b, segs, li, wide),
+        }
+        const got = try x.result(segs.len, x.b.kda.saves.out, out_bytes);
+        if (arm == .loop) {
+            want = got;
+            continue;
+        }
+        const what = try std.fmt.bufPrint(&name, "glm-seg-streams-check kda layer {d}, rows {any}, {s}, {s} {d}", .{ li, rows, if (wide) "three kernels" else "fused", @tagName(arm), if (arm == .streams) arm.streams else 1 });
+        same(t, what, got, want);
+    }
+    const st1 = &x.st[1];
+    try x.poke(1, st1.recAt(st1.cur[li], li) - st1.arena.buf.ptr + 401);
+    try x.reset(segs.len, x.b.kda.saves.out, out_bytes);
+    try seg.kdaChains(x.r.gpu.d, &x.ks, x.r.s, x.streamsOf(4), l, &x.b, segs, li, wide);
+    differs(t, try x.result(segs.len, x.b.kda.saves.out, out_bytes), want, try std.fmt.bufPrint(&name, "kda layer {d}, rows {any}", .{ li, rows }));
+}
+
+/// MLA inputs of layer di at `pos`: the latent rows, index keys and gates before it, the pools they complete.
+const MlaFill = struct {
+    di: usize,
+    pos: usize,
+
+    fn host(f: MlaFill, x: *Bench, st: *fw.State, img: []u8) !void {
+        st.pos = f.pos;
+        const p: i32 = @intCast(f.pos);
+        put(img, st, st.pos_dev, std.mem.asBytes(&p));
+        const c = st.caches[f.di];
+        if (c.ls == 0) put(img, st, c.lc, try bf16s(x.r, f.pos * 512, 1.0));
+        put(img, st, c.ik, try bf16s(x.r, f.pos * 128, 1.0));
+        put(img, st, c.ig, try bf16s(x.r, f.pos * 128, 2.0));
+        put(img, st, c.pk, try bf16s(x.r, f.pos / 4 * 128, 1.0));
+    }
+    /// An FP8 cache's rows through the writer itself: E4M3 codes and their scales.
+    fn device(f: MlaFill, x: *Bench, st: *fw.State) !void {
+        const c = st.caches[f.di];
+        if (c.ls == 0) return;
+        const zero: i32 = 0;
+        const rows = try x.r.dev(try bf16s(x.r, f.pos * 512, 1.0));
+        try glm.latent.latentWriteFormat(x.set, x.r.s, rows, 512, c.lc, c.ls, .{}, try x.r.dev(std.mem.asBytes(&zero)), f.pos, 512);
+    }
+};
+
+/// The MLA loop as the series wrote it before the knob, on the caller's stream.
+fn mlaLoop(x: *Bench, k: glm.dsa_block.Kernels, l: glm.dsa_block.Layer, segs: []const fw.Seg, di: usize) !void {
+    for (segs) |g| {
+        const nch = @min(glm.latent.chunksFor(g.st.pos + g.rows), x.b.nch);
+        try glm.dsa_block.attend(k, x.r.s, l, g.st.caches[di], x.b.dsa, g.row0, g.rows, g.st.pos_dev, g.st.pos, nch, x.b.part_rows);
+    }
+}
+
+/// One MLA case: every arm against the series' loop, then the red control (segment 1's first query row moved).
+fn mlaCase(x: *Bench, t: *Tally, rows: []const usize, at: []const usize, di: usize) !void {
+    const heads = x.w.share.mla.len();
+    var l: glm.dsa_block.Layer = undefined;
+    l.heads = heads;
+    l.lw = 512;
+    l.qk = 256;
+    l.dense_limit = x.w.config.keyWidth();
+    l.index = .{ .kw = undefined, .gate = 0, .ln_w = try x.r.dev(try bf16s(x.r, 128, 1.0)), .ln_b = try x.r.dev(try bf16s(x.r, 128, 0.1)), .ape = try x.r.dev(try bf16s(x.r, 4 * 128, 0.5)), .qb = undefined, .heads = 32, .d = 128 };
+    const k: glm.dsa_block.Kernels = .{ .set = x.set, .ops = x.ops };
+    var buf: [8]fw.Seg = undefined;
+    const segs = segments(x, rows, &buf);
+    var total: usize = 0;
+    for (rows) |m| total += m;
+    for (segs, at, 0..) |_, p, i| try x.start(i, MlaFill{ .di = di, .pos = p });
+    x.stage.clearRetainingCapacity();
+    try x.stageAt(x.b.dsa.lat, try bf16s(x.r, total * 512, 1.0));
+    try x.stageAt(x.b.dsa.ikr, try bf16s(x.r, total * 160, 1.0));
+    try x.stageAt(x.b.dsa.igr, try f32s(x.r, total * 128, 2.0));
+    try x.stageAt(x.b.dsa.qa, try bf16s(x.r, total * heads * 512, 0.5));
+    const out_bytes = total * heads * 512 * 2;
+    var want: []u8 = &.{};
+    var name: [200]u8 = undefined;
+    const fp8 = x.st[0].caches[di].ls != 0;
+    for ([_]Arm{ .loop, .in_order, .{ .streams = 2 }, .{ .streams = 4 }, .{ .streams = 8 } }) |arm| {
+        try x.reset(segs.len, x.b.dsa.ol, out_bytes);
+        switch (arm) {
+            .loop => try mlaLoop(x, k, l, segs, di),
+            .in_order => try seg.mlaAttends(k, x.r.s, null, l, &x.b, segs, di),
+            .streams => |n| try seg.mlaAttends(k, x.r.s, x.streamsOf(n), l, &x.b, segs, di),
+        }
+        const got = try x.result(segs.len, x.b.dsa.ol, out_bytes);
+        if (arm == .loop) {
+            want = got;
+            continue;
+        }
+        const what = try std.fmt.bufPrint(&name, "glm-seg-streams-check mla layer {d} ({s}), rows {any} at {any}, {s} {d}", .{ di, if (fp8) "fp8" else "bf16", rows, at, @tagName(arm), if (arm == .streams) arm.streams else 1 });
+        same(t, what, got, want);
+    }
+    const qa = x.stage.items[3]; // segment 1's first query row, one byte
+    const staged: cuda.DeviceBuffer = .{ .d = x.r.gpu.d, .ptr = qa[0], .len = qa[2] };
+    var one: [1]u8 = undefined;
+    try staged.download(rows[0] * heads * 512 * 2 + 7, &one);
+    one[0] ^= 0x40;
+    try staged.upload(rows[0] * heads * 512 * 2 + 7, &one);
+    try x.reset(segs.len, x.b.dsa.ol, out_bytes);
+    try seg.mlaAttends(k, x.r.s, x.streamsOf(4), l, &x.b, segs, di);
+    differs(t, try x.result(segs.len, x.b.dsa.ol, out_bytes), want, try std.fmt.bufPrint(&name, "mla layer {d}, rows {any}", .{ di, rows }));
+}
+
+pub fn run(gpu: check.Gpu) !void {
+    var arena: std.heap.ArenaAllocator = .init(gpu.gpa);
+    defer arena.deinit();
+    var s = try cuda.Stream.init(gpu.d, false);
+    defer s.deinit();
+    var prng = std.Random.DefaultPrng.init(0x5e95);
+    var r: Rig = .{ .gpu = gpu, .a = arena.allocator(), .s = s, .rng = prng.random() };
+    defer {
+        for (r.bufs.items) |*b| b.free();
+        r.bufs.deinit(gpu.gpa);
+    }
+    var mods: [3]cuda.Module = .{ try cuda.Module.load(gpu.d, cuda.kernels.glm_kda), try cuda.Module.load(gpu.d, cuda.kernels.glm_glue), try cuda.Module.load(gpu.d, cuda.kernels.glm_ops) };
+    defer for (&mods) |*m| m.unload();
+    const gks = try glm.glue.Kernels.loadOvn(mods[1], true);
+    var x: Bench = .{ .r = &r, .ks = try kda.Kernels.load(mods[0]), .set = .{ .glue = &gks, .mla = .{ .fuse_rows = glm.latent.prompt_rows, .skip = true } }, .ops = mods[2], .streams = undefined };
+    x.chunk = try r.a.alloc(u8, 16 << 20);
+    var made: usize = 0;
+    defer for (x.streams[0..made]) |*q| q.deinit();
+    for ([_]usize{ 2, 4, 8 }, 0..) |n, i| {
+        x.streams[i] = try seg.Streams.init(gpu.d, n);
+        made += 1;
+    }
+    for (&x.delay) |*p| p.* = try r.zeros(delay_bytes);
+    glm.kda_block.pad_to = glm.kda_block.padWidth(64, 3); // TF_GLM_KDA_PAD, as the recipe serves
+    defer glm.kda_block.pad_to = 0;
+    var t: Tally = .{};
+    // the bench's shapes: 2 to 8 streams' windows of 1 to 8 rows, the first, a middle and the last KDA layer
+    const kda_cases = [_]struct { rows: []const usize, li: usize }{
+        .{ .rows = &.{ 3, 4 }, .li = 0 },
+        .{ .rows = &.{ 2, 1, 4 }, .li = 17 },
+        .{ .rows = &.{ 3, 3, 2, 4 }, .li = 33 },
+        .{ .rows = &.{ 2, 5, 1, 3, 8, 2, 4, 3 }, .li = 9 },
+    };
+    // MLA at chunk edges and up to the dense limit (2051 keys), in the first, a middle and the last DSA layer
+    const mla_cases = [_]struct { rows: []const usize, at: []const usize, di: usize }{
+        .{ .rows = &.{ 2, 3 }, .at = &.{ 37, 509 }, .di = 0 },
+        .{ .rows = &.{ 4, 1, 3, 2 }, .at = &.{ 511, 1023, 1500, 2040 }, .di = 5 },
+        .{ .rows = &.{ 1, 2, 5, 3, 4, 2, 6, 1 }, .at = &.{ 5, 300, 512, 777, 1024, 1300, 1999, 2049 }, .di = 10 },
+    };
+    for ([_]bool{ false, true }) |kv8| {
+        try x.open(kv8);
+        defer x.close();
+        if (!kv8) for (kda_cases) |cs| for ([_]bool{ true, false }) |wide| try kdaCase(&x, &t, cs.rows, cs.li, wide);
+        for (mla_cases) |cs| try mlaCase(&x, &t, cs.rows, cs.at, cs.di);
+    }
+    std.debug.print("RESULT glm-seg-streams-check: {d} of {d} comparisons differ\n", .{ t.failed, t.cases });
+    try check.expect(t.failed == 0, "glm-seg-streams-check: {d} of {d} comparisons differ", .{ t.failed, t.cases });
+    check.pass("glm-seg-streams-check: KDA chains and MLA attention on 2, 4 and 8 streams == the in-order loop on {d} comparisons, red controls included", .{t.cases});
+}
